CREATE TABLE Utilisateur (
  id_user INT PRIMARY KEY AUTO_INCREMENT,
  nom VARCHAR(100) NOT NULL,
  email VARCHAR(100) UNIQUE NOT NULL,
  mot_de_passe VARCHAR(255) NOT NULL,
  telephone VARCHAR(20),
  role ENUM('client', 'admin') DEFAULT 'client'
);

CREATE TABLE Ville(
  id_ville INT PRIMARY KEY AUTO_INCREMENT,
  ville VARCHAR(100) NOT NULL,
  quartier VARCHAR(100) NOT NULL
);


CREATE TABLE Terrain (
  id_terrain INT PRIMARY KEY AUTO_INCREMENT,
  nom_terrain VARCHAR(100) NOT NULL,
  id_ville INT,
  prix_par_heure DECIMAL(10,2) NOT NULL,
  disponibilite BOOLEAN DEFAULT TRUE,
  FOREIGN KEY (id_ville) REFERENCES Ville(id_ville)
);


CREATE TABLE Reservation (
  id_reservation INT PRIMARY KEY AUTO_INCREMENT,
  id_user INT,
  id_terrain INT,
  date_reservation DATE NOT NULL,
  heure TIME NOT NULL,
  statut ENUM('confirmée','en_attente','annulée') DEFAULT 'en_attente',
  FOREIGN KEY (id_user) REFERENCES Utilisateur(id_user),
  FOREIGN KEY (id_terrain) REFERENCES Terrain(id_terrain)
);


CREATE TABLE Produit (
  id_produit INT PRIMARY KEY AUTO_INCREMENT,
  nom_produit VARCHAR(100) NOT NULL,
  description TEXT,
  prix DECIMAL(10,2) NOT NULL,
  stock INT NOT NULL, 
  image_url VARCHAR(255)
);

CREATE TABLE Commande (
  id_commande INT PRIMARY KEY AUTO_INCREMENT,
  id_user INT,
  date_commande DATE NOT NULL,
  statut ENUM('en_attente','envoyée','livrée') DEFAULT 'en_attente',
  FOREIGN KEY (id_user) REFERENCES Utilisateur(id_user)
);

CREATE TABLE Commande_Produit (
  id_commande INT,
  id_produit INT,
  quantite INT NOT NULL,
  PRIMARY KEY (id_commande, id_produit),
  FOREIGN KEY (id_commande) REFERENCES Commande(id_commande),
  FOREIGN KEY (id_produit) REFERENCES Produit(id_produit)
);

CREATE TABLE Paiement (
  id_paiement INT PRIMARY KEY AUTO_INCREMENT,
  id_user INT,
  montant DECIMAL(10,2) NOT NULL,
  methode ENUM('Carte Bancaire','PayPal','Maroc Telecommerce','Cash à la livraison'),
  date_paiement DATETIME NOT NULL,
  FOREIGN KEY (id_user) REFERENCES Utilisateur(id_user)
);


CREATE TABLE Commentaire_Terrain (
  id_commentaire INT PRIMARY KEY AUTO_INCREMENT,
  id_terrain INT,
  id_user INT,
  message TEXT NOT NULL,
  contact_whatsapp VARCHAR(20),
  date_commentaire DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (id_terrain) REFERENCES Terrain(id_terrain),
  FOREIGN KEY (id_user) REFERENCES Utilisateur(id_user)
);
